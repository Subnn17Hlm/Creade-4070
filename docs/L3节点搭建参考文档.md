### 函数签名
```python
def download_s3_file(s3_key: str, local_path: str) -> str:
    """
    从S3对象存储下载文件到本地。
    
    Args:
        s3_key: S3上的对象键（如 "materials/C0004_xxxx.mp4"）
        local_path: 本地保存路径（如 "/tmp/materials/C0004.mp4"）
    
    Returns:
        str: 下载后的本地文件路径

```

class SubtitleSegment(BaseModel):
    """单段字幕"""
    text: str = Field(..., description="字幕文本")

    l1_filter: Any                               # L1筛选条件（MVP暂未使用）
    l2_filter: Any                               # L2筛选条件（MVP暂未使用）
```

## 5. trim_node 当前素材匹配逻辑

### 所在文件
`src/graphs/nodes/trim_node.py`

### 当前逻辑（MVP模式——用户手动指定）

```python
# 1. 从用户传入的 material_assignments 中按分镜序号查找
material_assignments = state.material_assignments  # e.g. {"1": "C0004", "2": "C0026"}
assigned_material_id = material_assignments.get(str(shot_index))

if assigned_material_id:
    # 2. 从 material索引中取出该素材的ID
    material = material_map.get(assigned_material_id)
else:
    # 3. 未指定时：跳过该分镜
    logger.warning("分镜%d未指定素材，跳过", shot_index)
    continue
```

### 裁剪逻辑

```python
# 4. 以 TTS 实际语音时长 voice_duration 为绝对锚点
voice_duration = tts_durations[shot_index - 1]  # 由TTS节点计算

# 5. 从素材中间位置截取（素材足够长时取中点 ± 语音时长/2）
mid_point = material_duration / 2
half_duration = voice_duration / 2
start_time = max(0, mid_point - half_duration)
end_time = min(material_duration, start_time + voice_duration)

# 6. 素材不够长时使用全部时长（不做重复/慢放）
if material_duration <= voice_duration:
    start_time = 0
    end_time = material_duration

# 7. 质检指标（业务层记录/警告，不阻断流程）
usage_ratio = voice_duration / material_duration
if usage_ratio > 0.8:
    label = "素材过短"  # 素材几乎用满，无余量
elif usage_ratio < 0.5:
    label = "素材充裕"  # 素材只用了一小半
else:
    label = "匹配适中"
```

### L3节点的目标
**自动完成素材匹配：** 根据ShotItem的 `intent`、`selling_point_keywords`、`l2_filter`，从70个素材池中通过向量相似度或规则匹配，自动选出最优素材ID，替代当前的手动 `material_assignments` 输入。

def _call_llm_for_tagging(
    llm_client: LLMClient,   # 大模型客户端
    material_id: str,        # 素材ID（用于模板渲染）
    frame_urls: List[str],   # 帧图片URL列表
    sp: str,                 # 系统提示词（System Prompt）
    up_template: str,        # 用户提示词模板（Jinja2格式）
) -> Tuple[L1Tags, str]:

    调用视觉大模型对素材进行标注。
    返回: (标签对象, 标注来源)

    """
```

### 视觉模型调用完整代码示例

```python
from coze_coding_dev_sdk import LLMClient
from langchain_core.messages import SystemMessage, HumanMessage
from jinja2 import Template

# 1. 初始化视觉大模型（支持图片输入）
llm_client = LLMClient(
    model="doubao-seed-1-8-251228",  # 支持图文输入的多模态模型
    temperature=0.0,
)

# 2. 渲染用户提示词
up_tpl = Template(up_template)
user_prompt = up_tpl.render(material_id=material_id)

# 3. 构建图文混合消息
content_blocks = [{"type": "text", "text": user_prompt}]
for url in frame_urls:
    content_blocks.append({
        "type": "image_url",
        "image_url": {"url": url}
    })

messages = [
    SystemMessage(content=sp),
    HumanMessage(content=content_blocks),
]

# 4. 调用模型
response = llm_client.invoke(messages=messages)
text = response.content

# 5. 解析返回的JSON（可能含markdown代码块）
if isinstance(text, list):
    parts = []
    for item in text:
        if isinstance(item, dict) and item.get("type") == "text":
            parts.append(item.get("text", ""))
    raw = " ".join(parts)
else:
    raw = str(text)

# 提取JSON（兼容markdown ```json ... ``` 包裹）
if "```json" in raw:
    json_str = raw.split("```json")[1].split("```")[0].strip()
elif "```" in raw:
    json_str = raw.split("```")[1].split("```")[0].strip()
else:
    json_str = raw.strip()

data = json.loads(json_str)
```
